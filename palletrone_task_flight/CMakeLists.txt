cmake_minimum_required(VERSION 3.8)
project(palletrone_task_flight)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(std_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(dynamixel_sdk REQUIRED)
ament_python_install_package(${PROJECT_NAME})
add_executable(dynamixel_fix_with_j1 src/dynamixel_fix_with_j1.cpp)
target_compile_features(dynamixel_fix_with_j1 PUBLIC cxx_std_17)
target_compile_options(dynamixel_fix_with_j1 PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(dynamixel_fix_with_j1 PRIVATE include)
ament_target_dependencies(dynamixel_fix_with_j1 rclcpp rcl_interfaces std_msgs sensor_msgs dynamixel_sdk)
install(TARGETS dynamixel_fix_with_j1 DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/task_flight.py scripts/preview_task_flight.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(FILES README.md NOTICE.md LICENSE DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(task_geometry test/test_geometry.py)
  ament_add_pytest_test(task_state test/test_state_machine.py)
  ament_add_pytest_test(task_messages test/test_messages.py)
  add_executable(test_joint_math test/test_joint_math.cpp)
  target_compile_features(test_joint_math PUBLIC cxx_std_17)
  target_include_directories(test_joint_math PRIVATE include)
  add_test(NAME task_joint_math COMMAND test_joint_math)
endif()
ament_package()
